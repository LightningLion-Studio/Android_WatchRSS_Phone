cmake_minimum_required(VERSION 3.20)
project(QrAndroidJniHost LANGUAGES CXX)
find_package(JNI REQUIRED)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(ZXING_WRITERS OFF CACHE STRING "" FORCE)
set(ZXING_EXAMPLES OFF CACHE BOOL "" FORCE)
set(ZXING_C_API OFF CACHE BOOL "" FORCE)
set(ZXING_EXPERIMENTAL_API ON CACHE BOOL "" FORCE)
add_subdirectory("${ZXING_SOURCE}" zxing EXCLUDE_FROM_ALL)
add_library(zxingcpp_android SHARED "${ZXING_SOURCE}/wrappers/android/zxingcpp/src/main/cpp/ZXingCpp.cpp")
target_include_directories(zxingcpp_android PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include" ${JNI_INCLUDE_DIRS})
target_link_libraries(zxingcpp_android PRIVATE ZXing::ZXing)
